  1110→// Real-time collaboration over SSE + POST.
  1111→// Each session maintains a list of SSE subscribers. Edits arrive via POST and
  1112→// are broadcast to all subscribers. This avoids the need for a WebSocket
  1113→// dependency while still providing push-based real-time updates.
  1114→const collabSubscribers = new Map(); // sessionId -> Set<ServerResponse>
  1115→
  1116→collabRegistry.sessions.forEach = collabRegistry.sessions.forEach?.bind(collabRegistry.sessions);
  1117→const origCreate = CollabRegistry.prototype.create;
  1118→// Hook into session creation to wire up the broadcast on edit.
  1119→const origEdit = CollabSession.prototype.edit;
  1120→CollabSession.prototype.edit = function (peerId, operation) {
  1121→  const op = origEdit.call(this, peerId, operation);
  1122→  broadcastCollab(this.id, { type: "edit", ...op });
  1123→  return op;
  1124→};
  1125→CollabSession.prototype.updateCursor = function (peerId, cursor) {
  1126→  this.peers.get(peerId).cursor = cursor;
  1127→  this.emit("cursor", { peerId, ...cursor });
  1128→  broadcastCollab(this.id, { type: "cursor", peerId, cursor });
  1129→};

  1131→function broadcastCollab(sessionId, message) {
  1132→  const subs = collabSubscribers.get(sessionId);
  1133→  if (!subs) return;
  1134→  const payload = `data: ${JSON.stringify(message)}\n\n`;
  1135→  for (const sub of subs) {
  1136→    try { sub.write(payload); } catch {}
  1137→  }
  1138→}
