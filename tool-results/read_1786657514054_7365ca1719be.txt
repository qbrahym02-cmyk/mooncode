   141→function json(response, status, value, headers = {}) {
   142→  const body = JSON.stringify(value);
   143→  response.writeHead(status, { "content-type": "application/json; charset=utf-8", "content-length": Buffer.byteLength(body), "cache-control": "no-store", ...headers });
   144→  response.end(body);
   145→}

   147→async function body(request, maxBytes = 5_000_000) {
   148→  const chunks = [];
   149→  let size = 0;
   150→  for await (const chunk of request) {
   151→    size += chunk.length;
   152→    if (size > maxBytes) throw Object.assign(new Error("Request body is too large"), { status: 413 });
   153→    chunks.push(chunk);
   154→  }
   155→  if (!chunks.length) return {};
   156→  try { return JSON.parse(Buffer.concat(chunks).toString("utf8")); }
   157→  catch { throw Object.assign(new Error("Body must be valid JSON"), { status: 400 }); }
   158→}

   160→function publicPath(pathname) {
   161→  const requested = pathname === "/" ? "index.html" : pathname.replace(/^\/+/, "");
   162→  const target = path.resolve(publicDir, requested);
   163→  if (target !== publicDir && !target.startsWith(`${publicDir}${path.sep}`)) return null;
   164→  return target;
   165→}

   167→async function serveStatic(request, response, pathname) {
   168→  let target = publicPath(pathname);
   169→  if (!target) return false;
   170→  try {
   171→    let info = await stat(target);
   172→    if (info.isDirectory()) target = path.join(target, "index.html");
   173→    info = await stat(target);
   174→    const extension = path.extname(target).toLowerCase();
   175→    const content = await readFile(target);
   176→    response.writeHead(200, {
   177→      "content-type": mime[extension] || "application/octet-stream",
   178→      "content-length": content.length,
   179→      "cache-control": extension === ".html" ? "no-store" : "public, max-age=3600",
   180→      "x-content-type-options": "nosniff",
   181→      "content-security-policy": "default-src 'self'; img-src 'self' data: blob:; style-src 'self' 'unsafe-inline'; script-src 'self'; connect-src 'self'; frame-src 'self' blob:; object-src 'none'; base-uri 'none'; form-action 'self'",
   182→    });
   183→    response.end(content);
   184→    return true;
   185→  } catch (error) {
   186→    if (error?.code !== "ENOENT") throw error;
   187→    return false;
   188→  }
   189→}

   450→      return json(response, 415, { error: `Unsupported media type: ${fileMime}. Only image/* is accepted.` });

  1072→/**
  1073→ * Minimal multipart/form-data parser sufficient for image uploads. Extracts
  1074→ * the first file part and returns its filename, content-type, and buffer.
  1075→ * Returns null when the body cannot be parsed.
  1076→ */
  1077→function parseMultipart(buffer, contentType) {
  1078→  const boundaryMatch = contentType.match(/boundary=("?)([^";\s]+)\1/);
  1079→  if (!boundaryMatch) return null;
  1080→  const boundary = Buffer.from(`--${boundaryMatch[2]}`);
  1081→  const parts = [];
  1082→  let start = 0;
  1083→  while (true) {
  1084→    const idx = buffer.indexOf(boundary, start);
  1085→    if (idx < 0) break;
  1086→    if (start > 0) parts.push(buffer.slice(start, idx - 2)); // -2 for trailing CRLF
  1087→    start = idx + boundary.length + 2; // skip boundary + CRLF
  1088→  }
  1089→  for (const part of parts) {
  1090→    if (part.length === 0) continue;
  1091→    // Skip the closing boundary marker.
  1092→    if (part[0] === 0x2d && part[1] === 0x2d) continue; // starts with "--"
  1093→    const headerEnd = part.indexOf("\r\n\r\n");
  1094→    if (headerEnd < 0) continue;
  1095→    const headerText = part.slice(0, headerEnd).toString("utf8");
  1096→    const body = part.slice(headerEnd + 4);
  1097→    // Trailing CRLF before the next boundary.
  1098→    const crlf = Buffer.from("\r\n");
  1099→    const trimmed = body.length >= 2 && body[body.length - 2] === 0x0d && body[body.length - 1] === 0x0a ? body.slice(0, -2) : body;
  1100→    const dispositionMatch = headerText.match(/Content-Disposition:\s*form-data;[^\r\n]*filename="([^"]+)"/i);
  1101→    const typeMatch = headerText.match(/Content-Type:\s*([^\r\n]+)/i);
  1102→    if (dispositionMatch) {
  1103→      return {
  1104→        filename: dispositionMatch[1],
  1105→        contentType: typeMatch ? typeMatch[1].trim() : guessMimeFromName(dispositionMatch[1]),
  1106→        buffer: trimmed,
  1107→      };
  1108→    }
  1109→  }
  1110→  return null;
  1111→}

  1143→/**
  1144→ * Heuristic suggestion for the demo provider. Closes common patterns:
  1145→ * unclosed brackets, incomplete function signatures, console.log snippets.
  1146→ */
  1147→function computeHeuristicSuggestion(prefix, suffix, language) {
  1148→  const last = prefix.slice(-50);
  1149→  // Close unclosed brackets.
  1150→  const opens = (last.match(/[(\[{]/g) || []).length;
  1151→  const closes = (last.match(/[)\]}]/g) || []).length;
  1152→  if (opens > closes) {
  1153→    const stack = [];
  1154→    for (const ch of last) {
  1155→      if (ch === "(" || ch === "[" || ch === "{") stack.push(ch);
  1156→      else if (ch === ")" && stack[stack.length - 1] === "(") stack.pop();
  1157→      else if (ch === "]" && stack[stack.length - 1] === "[") stack.pop();
  1158→      else if (ch === "}" && stack[stack.length - 1] === "{") stack.pop();
  1159→    }
  1160→    const closeMap = { "(": ")", "[": "]", "{": "}" };
  1161→    return stack.reverse().map((ch) => closeMap[ch]).join("");
  1162→  }
  1163→  // Suggest function body opener.
  1164→  if (/\bfunction\s+\w+\s*\([^)]*\)\s*\{?\s*$/.test(last) && !last.endsWith("{")) {
  1165→    return "{\n  \n}";
  1166→  }
  1167→  // Suggest console.log for debugging.
  1168→  if (/\bconsole\b\.?$/.test(last)) return ".log()";
  1169→  if (/\breturn\s*$/.test(last)) return " ";
  1170→  return "";
  1171→}
