    47→export function estimateCost(model, usage) {
    48→  if (!usage) return null;
    49→  const inputTokens = Number(usage.inputTokens ?? usage.input_tokens ?? 0) || 0;
    50→  const outputTokens = Number(usage.outputTokens ?? usage.output_tokens ?? 0) || 0;
    51→  const price = PRICE_TABLE_USD_PER_1K[String(model).toLowerCase()];
    52→  if (!price) return { inputTokens, outputTokens, totalTokens: inputTokens + outputTokens, costUsd: null };
    53→  const costUsd = (inputTokens * price.input + outputTokens * price.output) / 1000;
    54→  return { inputTokens, outputTokens, totalTokens: inputTokens + outputTokens, costUsd };
    55→}

    91→/**
    92→ * Convert the internal message format to OpenAI's multimodal content schema.
    93→ * Internal messages can be either `{ role, content: string }` or
    94→ * `{ role, content: [{ type: "text", text }, { type: "image_url", image_url: { url } }] }`.
    95→ * String content passes through unchanged.
    96→ */
    97→function toOpenAIMessages(messages = []) {
    98→  return messages.map((message) => {
    99→    if (typeof message.content === "string") return message;
   100→    if (Array.isArray(message.content)) {
   101→      return { ...message, content: message.content.map((part) => {
   102→        if (part.type === "text") return { type: "text", text: part.text };
   103→        if (part.type === "image_url") return { type: "image_url", image_url: { url: part.image_url?.url ?? part.url } };
   104→        return part;
   105→      }) };
   106→    }
   107→    return message;
   108→  });
   109→}

   111→function toAnthropicMessages(messages = []) {
   112→  return messages.map((message) => {
   113→    if (typeof message.content === "string") {
   114→      return { ...message, content: typeof message.content === "string" ? message.content : JSON.stringify(message.content) };
   115→    }
   116→    if (Array.isArray(message.content)) {
   117→      const parts = message.content.map((part) => {
   118→        if (part.type === "text") return { type: "text", text: part.text };
   119→        if (part.type === "image_url") {
   120→          const url = part.image_url?.url ?? part.url ?? "";
   121→          // Anthropic expects base64-encoded image data with explicit source.
   122→          const match = url.match(/^data:(image\/[a-zA-Z+]+);base64,(.+)$/);
   123→          if (match) {
   124→            return { type: "image", source: { type: "base64", media_type: match[1], data: match[2] } };
   125→          }
   126→          return { type: "text", text: `(image: ${url.slice(0, 80)})` };
   127→        }
   128→        return part;
   129→      });
   130→      return { ...message, content: parts };
   131→    }
   132→    return message;
   133→  });
   134→}

   136→function toGoogleMessages(messages = []) {
   137→  return messages.map((message) => {
   138→    if (typeof message.content === "string") {
   139→      return { role: message.role === "assistant" ? "model" : "user", parts: [{ text: message.content }] };
   140→    }
   141→    if (Array.isArray(message.content)) {
   142→      const parts = message.content.map((part) => {
   143→        if (part.type === "text") return { text: part.text };
   144→        if (part.type === "image_url") {
   145→          const url = part.image_url?.url ?? part.url ?? "";
   146→          const match = url.match(/^data:(image\/(png|jpeg|webp|gif));base64,(.+)$/);
   147→          if (match) {
   148→            return { inlineData: { mimeType: match[1], data: match[3] } };
   149→          }
   150→          return { text: `(image: ${url.slice(0, 80)})` };
   151→        }
   152→        return { text: JSON.stringify(part) };
   153→      });
   154→      return { role: message.role === "assistant" ? "model" : "user", parts };
   155→    }
   156→    return { role: message.role === "assistant" ? "model" : "user", parts: [{ text: String(message.content ?? "") }] };
   157→  });
   158→}

   160→async function callOpenAI(config, request) {
   161→  const baseUrl = String(config.baseUrl).replace(/\/$/, "");
   162→  const body = {
   163→    model: config.model,
   164→    messages: toOpenAIMessages(request.messages),
   165→    temperature: request.temperature ?? 0.2,
   166→    stream: Boolean(request.stream),
   167→    ...(request.tools?.length ? { tools: openAITools(request.tools), tool_choice: "auto" } : {}),
   168→  };
   169→  if (!request.stream) {
   170→    const data = await jsonRequest(`${baseUrl}/chat/completions`, {
   171→      method: "POST",
   172→      headers: {
   173→        "content-type": "application/json",
   174→        ...(config.apiKey ? { authorization: `Bearer ${config.apiKey}` } : {}),
   175→        ...(config.headers ?? {}),
   176→      },
   177→      body: JSON.stringify(body),
   178→    });
   179→    return normalizeOpenAI(data);
   180→  }
   181→  // SSE streaming path
   182→  const response = await fetch(`${baseUrl}/chat/completions`, {
   183→    method: "POST",
   184→    headers: {
   185→      "content-type": "application/json",
   186→      accept: "text/event-stream",
   187→      ...(config.apiKey ? { authorization: `Bearer ${config.apiKey}` } : {}),
   188→      ...(config.headers ?? {}),
   189→    },
   190→    body: JSON.stringify(body),
   191→    signal: request.signal ?? withTimeout(),
   192→  });
   193→  if (!response.ok) {
   194→    const text = await response.text();
   195→    let detail = null;
   196→    try { detail = JSON.parse(text); } catch { detail = { raw: text.slice(0, 4_000) }; }
   197→    throw new ProviderError(detail?.error?.message || `Provider request failed (${response.status})`, response.status, detail);
   198→  }
   199→  const reader = response.body.getReader();
   200→  const decoder = new TextDecoder();
   201→  let buffer = "";
   202→  let usage = null;
   203→  let stopReason = null;
   204→  const toolCalls = new Map();
   205→  let textBuffer = "";
   206→  const ensureToolCall = (id, name) => {
   207→    if (!toolCalls.has(id)) toolCalls.set(id, { id, name, arguments: "" });
   208→    return toolCalls.get(id);
   209→  };
   210→  while (true) {
   211→    const { value, done } = await reader.read();
   212→    if (done) break;
   213→    buffer += decoder.decode(value || new Uint8Array(), { stream: true });
   214→    const lines = buffer.split("\n");
   215→    buffer = lines.pop() || "";
   216→    for (const line of lines) {
   217→      const trimmed = line.trim();
   218→      if (!trimmed.startsWith("data:")) continue;
   219→      const payload = trimmed.slice(5).trim();
   220→      if (payload === "[DONE]") continue;
   221→      let event;
   222→      try { event = JSON.parse(payload); } catch { continue; }
   223→      if (event.usage) usage = { inputTokens: event.usage.prompt_tokens, outputTokens: event.usage.completion_tokens };
   224→      const choice = event.choices?.[0];
   225→      if (!choice) continue;
   226→      if (choice.finish_reason) stopReason = choice.finish_reason;
   227→      const delta = choice.delta ?? {};
   228→      if (delta.content) {
   229→        textBuffer += delta.content;
   230→        request.onDelta?.(delta.content);
   231→      }
   232→      if (Array.isArray(delta.tool_calls)) {
   233→        for (const partial of delta.tool_calls) {
   234→          const id = partial.id ?? `call_${toolCalls.size}`;
   235→          const entry = ensureToolCall(id, partial.function?.name);
   236→          if (partial.function?.name && !entry.name) entry.name = partial.function.name;
   237→          if (partial.function?.arguments) entry.arguments += partial.function.arguments;
   238→        }
   239→      }
   240→    }
   241→  }
   242→  const assembled = [...toolCalls.values()].map((entry) => {
   243→    let input = {};
   244→    try { input = JSON.parse(entry.arguments || "{}"); } catch { input = { _raw: entry.arguments }; }
   245→    return { id: entry.id, name: entry.name, input };
   246→  });
   247→  return { text: textBuffer, stopReason: stopReason || "stop", usage, toolCalls: assembled };
   248→}

   250→async function callAnthropic(config, request) {
   251→  const system = request.messages.filter((m) => m.role === "system").map((m) => typeof m.content === "string" ? m.content : (m.content?.filter((p) => p.type === "text").map((p) => p.text).join("\n") ?? "")).join("\n\n");
   252→  const messages = toAnthropicMessages(request.messages.filter((m) => m.role !== "system"));
   253→  const body = {
   254→    model: config.model,
   255→    max_tokens: request.maxTokens ?? 8_192,
   256→    system,
   257→    messages,
   258→    stream: Boolean(request.stream),
   259→    ...(request.tools?.length ? { tools: request.tools.map((tool) => ({ name: tool.name, description: tool.description, input_schema: tool.inputSchema })) } : {}),
   260→  };
   261→  if (!request.stream) {
   262→    const data = await jsonRequest(`${String(config.baseUrl).replace(/\/$/, "")}/messages`, {
   263→      method: "POST",
   264→      headers: { "content-type": "application/json", "x-api-key": config.apiKey, "anthropic-version": "2023-06-01" },
   265→      body: JSON.stringify(body),
   266→    });
   267→    return {
   268→      text: (data.content ?? []).filter((part) => part.type === "text").map((part) => part.text).join("\n"),
   269→      toolCalls: (data.content ?? []).filter((part) => part.type === "tool_use").map((part) => ({ id: part.id, name: part.name, input: part.input ?? {} })),
   270→      stopReason: data.stop_reason,
   271→      usage: data.usage ? { inputTokens: data.usage.input_tokens, outputTokens: data.usage.output_tokens } : null,
   272→    };
   273→  }
   274→  // SSE streaming path
   275→  const response = await fetch(`${String(config.baseUrl).replace(/\/$/, "")}/messages`, {
   276→    method: "POST",
   277→    headers: { "content-type": "application/json", "x-api-key": config.apiKey, "anthropic-version": "2023-06-01", accept: "text/event-stream" },
   278→    body: JSON.stringify(body),
   279→    signal: request.signal ?? withTimeout(),
   280→  });
   281→  if (!response.ok) {
   282→    const text = await response.text();
   283→    let detail = null;
   284→    try { detail = JSON.parse(text); } catch { detail = { raw: text.slice(0, 4_000) }; }
   285→    throw new ProviderError(detail?.error?.message || `Provider request failed (${response.status})`, response.status, detail);
   286→  }
   287→  const reader = response.body.getReader();
   288→  const decoder = new TextDecoder();
   289→  let buffer = "";
   290→  let textBuffer = "";
   291→  let usage = null;
   292→  let stopReason = null;
   293→  const toolCalls = new Map();
   294→  while (true) {
   295→    const { value, done } = await reader.read();
   296→    if (done) break;
   297→    buffer += decoder.decode(value || new Uint8Array(), { stream: true });
   298→    const events = buffer.split("\n\n");
   299→    buffer = events.pop() || "";
   300→    for (const block of events) {
   301→      const lines = block.split("\n");
   302→      const dataLine = lines.find((line) => line.startsWith("data:"));
   303→      const eventLine = lines.find((line) => line.startsWith("event:"));
   304→      if (!dataLine) continue;
   305→      const eventType = eventLine ? eventLine.slice(6).trim() : "message";
   306→      let payload;
   307→      try { payload = JSON.parse(dataLine.slice(5).trim()); } catch { continue; }
   308→      if (payload.type) {
   309→        // Anthropic sends `type` inside data
   310→        if (payload.type === "content_block_delta" && payload.delta?.text) {
   311→          textBuffer += payload.delta.text;
   312→          request.onDelta?.(payload.delta.text);
   313→        } else if (payload.type === "content_block_start" && payload.content_block?.type === "tool_use") {
   314→          toolCalls.set(payload.index, { id: payload.content_block.id, name: payload.content_block.name, arguments: "" });
   315→        } else if (payload.type === "content_block_delta" && payload.delta?.type === "input_json_delta") {
   316→          const entry = toolCalls.get(payload.index);
   317→          if (entry) entry.arguments += payload.delta.partial_json;
   318→        } else if (payload.type === "message_delta") {
   319→          if (payload.delta?.stop_reason) stopReason = payload.delta.stop_reason;
   320→          if (payload.usage) usage = { inputTokens: usage?.inputTokens ?? 0, outputTokens: payload.usage.output_tokens ?? usage?.outputTokens ?? 0 };
   321→        } else if (payload.type === "message_start" && payload.message?.usage) {
   322→          usage = { inputTokens: payload.message.usage.input_tokens, outputTokens: payload.message.usage.output_tokens };
   323→        }
   324→      } else if (eventType === "message_stop") {
   325→        // No-op
   326→      }
   327→    }
   328→  }
   329→  const assembled = [...toolCalls.values()].map((entry) => {
   330→    let input = {};
   331→    try { input = JSON.parse(entry.arguments || "{}"); } catch { input = { _raw: entry.arguments }; }
   332→    return { id: entry.id, name: entry.name, input };
   333→  });
   334→  return { text: textBuffer, stopReason: stopReason || "stop", usage, toolCalls: assembled };
   335→}

   434→  const decoder = new TextDecoder();
   435→  let buffer = "";
   436→  let textBuffer = "";
   437→  let usage = null;
   438→  let stopReason = null;
   439→  const toolCalls = [];
   440→  while (true) {
   441→    const { value, done } = await reader.read();
   442→    if (done) break;
   443→    buffer += decoder.decode(value || new Uint8Array(), { stream: true });
   444→    const lines = buffer.split("\n");
   445→    buffer = lines.pop() || "";
   446→    for (const line of lines) {
   447→      if (!line.trim()) continue;
   448→      let event;
   449→      try { event = JSON.parse(line); } catch { continue; }
   450→      if (event.message?.content) {
   451→        textBuffer += event.message.content;
   452→        request.onDelta?.(event.message.content);
   453→      }
   454→      if (Array.isArray(event.message?.tool_calls)) {
   455→        for (const call of event.message.tool_calls) toolCalls.push({ id: crypto.randomUUID(), name: call.function?.name, input: call.function?.arguments ?? {} });
   456→      }
   457→      if (event.done) {
   458→        stopReason = event.done_reason ?? "stop";
   459→        usage = { inputTokens: event.prompt_eval_count, outputTokens: event.eval_count };
   460→      }
   461→    }
   462→  }
   463→  return { text: textBuffer, stopReason: stopReason || "stop", usage, toolCalls: toolCalls };
   464→}

   535→export async function callModel(options, request, env = process.env) {
   536→  const preset = PROVIDERS[options.provider] ?? PROVIDERS.custom;
   537→  const config = {
   538→    ...preset,
   539→    model: options.model || preset.defaultModel,
   540→    baseUrl: options.baseUrl || (options.provider === "custom" ? env.CUSTOM_BASE_URL : preset.baseUrl),
   541→    apiKey: options.apiKey || (preset.env ? env[preset.env] : undefined),
   542→    headers: options.headers,
   543→  };
   544→  if (preset.kind === "demo") return demo(request);
   545→  if (!config.baseUrl) throw new ProviderError("Provider base URL is required", 400);
   546→  if (preset.kind !== "ollama" && options.provider !== "custom" && !config.apiKey) throw new ProviderError(`API key is not configured for ${preset.label}`, 400);
   547→  if (preset.kind === "anthropic") return callAnthropic(config, request);
   548→  if (preset.kind === "google") return callGoogle(config, request);
   549→  if (preset.kind === "ollama") return callOllama(config, request);
   550→  return callOpenAI(config, request);
   551→}
